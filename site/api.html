import { xpict } from "@swamidasslab/xpict";

const mol = xpict.mol("c1ccccc1");
const rendered = await xpict.render(mol, {
  mark_atoms: [0],
  color: "#0b6e4f",
});
const svg = xpict.toSvg(rendered.scene);

const aligned = await xpict.render(xpict.mol("Cc1ccccc1"), {
  align_to: mol,
});

const batch = await xpict.depict({
  molecules: [
    { smiles: "CCO", mark_atoms: [2] },
    { smiles: "CCCO", align_to: 0 },
  ],
});

use xpict::{depict, mol, DepictSpec, MolRenderOptions, MolSpec};

let mut m = mol("CCO")?;
let rendered = m.render(MolRenderOptions {
    mark_atoms: Some(vec![2]),
    ..Default::default()
})?;
let svg = rendered.to_svg();

let batch = depict(&DepictSpec {
    molecules: vec![
        MolSpec {
            smiles: Some("CCO".into()),
            mark_atoms: Some(vec![2]),
            ..Default::default()
        },
        MolSpec {
            smiles: Some("CCCO".into()),
            align_to: Some(0),
            ..Default::default()
        },
    ],
})?;

Render options

Option Effect
color Backbone / label ink
mark_atoms / mark_bonds Publication circles
atom_shade / bond_shade Plot-dot shading scores
star_labels Labels for * (encounter order); else CXSMILES |$…$| by index
bold_labels Bold Liberation + stem-keyed stroke
align_to Template pose (Mol/Rendered in JS·Py; molblock / batch index in Rust·depict)
